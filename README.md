# Лабораторная работа №2: Анализ сложности алгоритмов и шаблоны функций

*   **Задача 2:** Реализовать шаблонную функцию `void swap(T& a, T& b)` (обмен значений двух переменных).
*   **Задача 3:** Продемонстрировать работу шаблонной функции с разными типами данных.

## 📚 Теоретическая часть
**О-нотация (Big O)** — это математическая запись, описывающая асимптотическое поведение функции (как быстро растет время работы алгоритма при увеличении входных данных). 
*   **O(1)** — константная (доступ к массиву).
*   **O(log n)** — логарифмическая (бинарный поиск).
*   **O(n)** — линейная (один цикл).
*   **O(n log n)** — линейно-логарифмическая (быстрая сортировка).
*   **O(n²)** — квадратичная (два вложенных цикла).

**Шаблоны функций** — механизм C++, позволяющий писать обобщенный код. Вместо множества перегрузок для `int`, `double`, `string` мы пишем один шаблон `template <typename T>`, а компилятор сам подставляет нужный тип.

## 💻 Листинг программы (main.cpp)

```cpp
#include <iostream>
#include <string>

using namespace std;

// ==========================================
// Шаблонная функция swap (Вариант 4)
// ==========================================
template <typename T>
void mySwap(T& a, T& b) {
    T temp = a;
    a = b;
    b = temp;
}

// ==========================================
// Фрагменты кода для анализа сложности
// ==========================================

// Пример А: O(n) - линейный проход
int exampleA(int arr[], int n) {
    int sum = 0;
    for (int i = 0; i < n; i++) { // n итераций
        sum += arr[i];
    }
    return sum;
}

// Пример B: O(n²) - два вложенных цикла
int exampleB(int arr[], int n) {
    int count = 0;
    for (int i = 0; i < n; i++) {     // n итераций
        for (int j = 0; j < n; j++) { // n итераций
            if (arr[i] == arr[j]) count++;
        }
    }
    return count;
}

// Пример C: O(log n) - логарифмическая
int exampleC(int n) {
    int count = 0;
    for (int i = 1; i < n; i *= 2) { // i удваивается: log2(n) итераций
        count++;
    }
    return count;
}

// Пример D: O(n log n)
int exampleD(int n) {
    int count = 0;
    for (int i = 0; i < n; i++) {     // n итераций
        for (int j = n; j > 0; j /= 2) { // log2(n) итераций
            count++;
        }
    }
    return count;
}

// Пример E: O(√n)
int exampleE(int n) {
    int count = 0;
    for (int i = 0; i * i < n; i++) { // i до √n
        count++;
    }
    return count;
}

// ==========================================
// Главная функция
// ==========================================
int main() {
    setlocale(LC_ALL, "ru");

    cout << "===========================================" << endl;
    cout << "1. Демонстрация работы шаблонной функции swap" << endl;
    cout << "===========================================" << endl;

    // Демонстрация с int
    int x = 5, y = 10;
    cout << "До swap (int): x = " << x << ", y = " << y << endl;
    mySwap(x, y);
    cout << "После swap (int): x = " << x << ", y = " << y << endl << endl;

    // Демонстрация с double
    double d1 = 3.14, d2 = 2.71;
    cout << "До swap (double): d1 = " << d1 << ", d2 = " << d2 << endl;
    mySwap(d1, d2);
    cout << "После swap (double): d1 = " << d1 << ", d2 = " << d2 << endl << endl;

    // Демонстрация с string
    string s1 = "Hello", s2 = "World";
    cout << "До swap (string): s1 = " << s1 << ", s2 = " << s2 << endl;
    mySwap(s1, s2);
    cout << "После swap (string): s1 = " << s1 << ", s2 = " << s2 << endl << endl;

    // Демонстрация с char
    char c1 = 'A', c2 = 'B';
    cout << "До swap (char): c1 = " << c1 << ", c2 = " << c2 << endl;
    mySwap(c1, c2);
    cout << "После swap (char): c1 = " << c1 << ", c2 = " << c2 << endl << endl;

    cout << "===========================================" << endl;
    cout << "2. Анализ сложности фрагментов кода" << endl;
    cout << "===========================================" << endl;
    
    int testArr[] = {1, 2, 3, 4, 5};
    exampleA(testArr, 5);
    exampleB(testArr, 5);
    exampleC(100);
    exampleD(10);
    exampleE(100);

    cout << "Пример A: O(n)      - один цикл, зависящий от n (линейный проход)." << endl;
    cout << "Пример B: O(n^2)    - два вложенных цикла, каждый выполняется n раз." << endl;
    cout << "Пример C: O(log n)  - шаг цикла умножается на 2 (i *= 2), итераций log(n)." << endl;
    cout << "Пример D: O(n log n)- внешний цикл n раз, внутренний цикл log(n) раз (j /= 2)." << endl;
    cout << "Пример E: O(sqrt(n))- условие i * i < n, цикл выполняется корень из n раз." << endl;

    return 0;
}
